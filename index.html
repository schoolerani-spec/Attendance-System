function listenToAttendanceLogs() { if (!currentUser) return; if (unsubscribeLogs) unsubscribeLogs(); unsubscribeLogs = db.collection("users") .doc(currentUser.uid) .collection("attendance_logs") .orderBy("timestamp", "desc") .limit(50) .onSnapshot(snapshot => { logTable.innerHTML = ""; let lastSeenDate = null; snapshot.forEach(doc => { const log = doc.data(); const dateStr = log.date || new Date().toLocaleDateString(); // Insert date divider row whenever the date changes if (dateStr !== lastSeenDate) { renderDateDividerRow(dateStr); lastSeenDate = dateStr; } renderLogEntry(log.name, log.time, log.statusText, log.badgeClass); }); }); } function renderDateDividerRow(dateStr) { const dividerRow = document.createElement('tr'); dividerRow.innerHTML = ` 📅 ${dateStr} `; logTable.appendChild(dividerRow); } function renderLogEntry(name, timeStr, statusText, badgeClass) { const row = document.createElement('tr'); row.innerHTML = ` ${name} - ${timeStr} ${statusText} `; logTable.appendChild(row); }
